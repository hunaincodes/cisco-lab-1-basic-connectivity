# 1. Git repository initialize karein
git init

# 2. Saari files ko staging area mein add karein
git add .

# 3. First commit karein
git commit -m "Add Lab 1 Packet Tracer file, screenshot and README documentation"

# 4. Main branch set karein
git branch -M main

# 5. Apni GitHub repo ko link karein (Apna GitHub URL copy karke yahan paste karein)
git remote add origin https://github.com/YOUR-USERNAME/cisco-lab-1-basic-connectivity.git

# 6. Files push karein
git push -u origin main
